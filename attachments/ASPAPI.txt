def clean_message(text: str) -> str:
    if not text:
        return ""

    # 1. Remove URLs, mentions, hashtags
    text = re.sub(r"https?://\S+|www\.\S+|@\w+|#\w+", "", text)

    # 2. ULTRA-AGGRESSIVE KILL OF EVERYTHING AFTER BLOCKBEATS OR "ACCORDING TO"
    kill_triggers = [
        r"BlockBeats",                  # Any mention of BlockBeats
        r"According\s+to\s+BlockBeats",  # "According to BlockBeats"
        r"According\s+to",              # Any "According to..." (catches Binance, official, etc.)
        r"Source\s*:\s*BlockBeats",     # Source: BlockBeats
        r"via\s+BlockBeats",            # via BlockBeats
    ]

    for trigger in kill_triggers:
        match = re.search(trigger, text, flags=re.IGNORECASE)
        if match:
            text = text[:match.start()].strip()
            logger.debug(f"Killed footer at: {trigger}")
            break  # Stop after first match

    # 3. Additional fallback: kill any leftover Bloomberg/BlockBeats lines
    lines = []
    for line in text.split('\n'):
        line = line.strip()
        lower = line.lower()
        if any(bad in lower for bad in [
            "blockbeats", "bloomberg", "according to", "source:", "via ", 
            "thea monitoring", "auntie ai", "on november", "on december"
        ]):
            continue
        if line:
            lines.append(line)

    text = "\n".join(lines)

    # 4. Remove junk words
    junk = ["Subscribe", "Follow"]
    for w in junk:
        text = re.sub(rf"\b{w}\b", "", text, flags=re.IGNORECASE)

    # 5. Final cleanup
    text = re.sub(r"\s+", " ", text)
    text = re.sub(r"\.+$", "", text)  # Remove trailing dots
    text = text.strip()

    return text if text else ""

def is_weird_message(text: str) -> bool:
    if not text or len(text.strip()) < 8:
        return True
    stripped = text.strip()
    lower = stripped.lower()

    # Pure number / price spam
    if re.match(r'^\s*[\d$,.\s!]+\s*$', stripped):
        return True

    # Classic BlockBeats bug patterns
    if re.search(r'now\s+\$\d+', lower) and len(stripped.split()) <= 7:
        return True
    if re.search(r'bitcoin now \$\d+', lower):
        return True
    if re.search(r'^\s*\d{4,6}\s*[!?.]*\s*$', lower):  # just "45000" or "45000!"
        return True

    # Too short or ends with !
    if len(stripped) < 25 and stripped.endswith('!'):
        return True

    return False

def store_message_in_file(text):
    """Store a message in ASPstorage.txt with a UTC timestamp."""
    try:
        # Generate UTC timestamp in format [YYYY-MM-DD HH:MM:SS]
        timestamp = datetime.now(timezone.utc).strftime('[%Y-%m-%d %H:%M:%S]')
        # Format message with timestamp
        formatted_message = f"{timestamp} {text}"
        
        with open(CRYPTO_FINANCIAL_FILE, 'a', encoding='utf-8') as f:
            f.write(formatted_message + "\n")
        logger.debug(f"Stored message in {CRYPTO_FINANCIAL_FILE}: '{formatted_message}'")
    except Exception as e:
        logger.error(f"Error storing message in file: {e}")

async def fetch_with_retry(session, url, params=None, max_retries=3, base_timeout=20):
    """SUPER ROBUST INFINITE RETRY — never gives up on internet disconnect, slow connection, or server fault.
    Keeps retrying forever with smart exponential backoff until the connection/server is restored.
    The script will NEVER fail or stop."""
    attempt = 0
    delay = 2  # start with 2 seconds

    while True:
        attempt += 1
        try:
            # Progressive timeout (gets more generous on repeated attempts)
            timeout = ClientTimeout(
                total=base_timeout + (attempt * 5),   # 25s → 30s → 35s ...
                sock_connect=15,
                sock_read=30
            )

            async with session.get(
                url,
                params=params,
                timeout=timeout,
                ssl=False,
                headers={"User-Agent": "Mozilla/5.0 (compatible; ASPNewsBot/1.0)"}
            ) as resp:

                if resp.status != 200:
                    raise ClientResponseError(resp.request_info, resp.history, status=resp.status)

                data = await resp.json()

                if attempt > 1:
                    logger.warning("━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
                    logger.warning(f"✅ CONNECTION RESTORED")
                    logger.warning(f"Source: {url}")
                    logger.warning(f"Recovered after {attempt} attempts")
                    logger.warning("━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
                return data

        except (asyncio.TimeoutError, ClientConnectorError, ServerTimeoutError, OSError,
                ClientResponseError, ConnectionError, TimeoutError) as e:
            logger.warning(f"🌐 Connection/Server issue on {url} (attempt {attempt}) — {type(e).__name__}: {str(e)[:150]}")
            logger.warning(f"   Retrying in {delay} seconds... (will keep trying forever until restored)")

            await asyncio.sleep(delay)
            delay = min(delay * 2, 300)   # backoff: 2s → 4s → 8s ... max 5 minutes
            continue

        except Exception as e:
            # Any other unexpected error — still retry forever (never crash)
            logger.error(f"💥 Unexpected error on {url} (attempt {attempt}): {type(e).__name__} - {str(e)[:200]}")
            await asyncio.sleep(delay)
            delay = min(delay * 2, 300)
            continue

# ── AI rewrite ────────────────────────────────────────────────────────
async def ai_rewrite(final_caption: str) -> str:
    if not final_caption.strip() or len(final_caption) < 20:
        return final_caption

    original = final_caption

    for provider in PROVIDERS:
        try:
            kwargs = {
                "model": provider["model"],
                "messages": [
                    {"role": "system", "content": REWRITE_SYSTEM_PROMPT},
                    {"role": "user",   "content": final_caption},
                ],
                "temperature": 0.15,          # lower = more stable
                "max_tokens": 400,            # a bit more room
                "top_p": 0.9,
                "timeout": 10,
            }

            if provider["type"] == "openai":
                kwargs["api_key"] = provider["api_key"]
                kwargs["api_base"] = provider.get("base_url")
            elif provider["type"] == "gemini":
                kwargs["api_key"] = provider["api_key"]

            response = await asyncio.to_thread(
                completion,
                **kwargs
            )
            rewritten = response.choices[0].message.content.strip()

            # Clean markdown if any
            if "```" in rewritten:
                code_content = re.search(r'```(?:\w+)?\s*(.*?)\s*```', rewritten, re.DOTALL)
                if code_content:
                    rewritten = code_content.group(1).strip()

            # === QUALITY CHECKS (this is the important part) ===
            if not rewritten or len(rewritten) < 20:
                logger.warning(f"{provider['name']} returned too short → skip")
                continue

            # Reject if it is drastically shorter than original (truncation detection)
            if len(rewritten) < len(original) * 0.55:
                logger.warning(f"{provider['name']} truncated output → skip")
                continue

            # Reject if it ends with incomplete words / mid-sentence
            if re.search(r'[a-zA-Z]{3,}$', rewritten) and not rewritten.endswith(('.', '!', '?', '"', "'")):
                # ends with a word but no proper punctuation → likely truncated
                if len(rewritten.split()) < len(original.split()) * 0.7:
                    logger.warning(f"{provider['name']} incomplete sentence → skip")
                    continue

            # Preserve leading tag if the model removed it
            tag_match = re.match(r"^(JUST IN:|BREAKING:|ALERT:|NEW:)\s*", original, re.I)
            if tag_match:
                tag = tag_match.group(1)
                if not rewritten.upper().startswith(tag.upper()):
                    rewritten = tag + " " + rewritten.lstrip(tag).lstrip(": ").lstrip()

            logger.info(f"AI rewrite success via {provider['name']}")
            return rewritten

        except Exception as e:
            logger.warning(f"{provider['name']} failed: {str(e)[:150]}")
            continue

    logger.error("All AI providers failed → using original caption")
    return original

async def process_and_forward_text(raw_text: str, unique_id: str, original_message=None):
    global last_id
    if unique_id in seen_ids:
        return
    if not raw_text or not raw_text.strip():
        return

    # === TRANSLATE FIRST ===
    text_to_use = raw_text.strip()
    try:
        translated = MyMemoryTranslator(source='zh-CN', target='en-GB').translate(text_to_use)
        text_to_use = translated
        logger.debug(f"✅ Translated: {text_to_use[:150]}...")
    except Exception as e:
        logger.warning(f"Translation failed: {e}")
        text_to_use = raw_text.strip()

    cleaned = clean_message(text_to_use)
    if not cleaned.strip():
        return

    if is_weird_message(cleaned):
        logger.info(f"TG WEIRD BLOCKED → {cleaned[:80]}...")
        seen_ids.add(unique_id)
        return
    
    # === PRE-AI BLOCKS - PROHIBITED PHRASES ===
    BLOCKED_PHRASES = [
        r"-\s*Link",                    # ← Active: This is the main one you want
        
        # === Chinese patterns (for original messages) ===
        # r"观点：",
        # r"作者观点",
        # r"仅供参考",
        # r"风险提示",
        
        # === English patterns (after translation) ===
        # r"korean", 
        # r"viewpoint:",
        # r"for reference only",
        # r"this is just an opinion",
        # r"not financial advice",
    ]

    if any(re.search(pattern, cleaned, re.IGNORECASE) for pattern in BLOCKED_PHRASES):
        logger.info(f"TG BLOCKED PROHIBITED PHRASE → {cleaned[:120]}...")
        seen_ids.add(unique_id)
        return

    # === NEW: AGGRESSIVE LONG MESSAGE BLOCKER ===
    word_count = len(cleaned.split())
    text_length = len(cleaned)
    
    if text_length > 160 or word_count > 30:
        logger.info(f"⛔ LONG MESSAGE BLOCKED ({word_count} words, {text_length} chars) → {cleaned[:100]}...")
        seen_ids.add(unique_id)
        return

    # === PRE-AI BLOCKS ===
    BLOCKED_WORDS = ["buddy", "trader", "orders", "listing", "airdrop", "meme", "memecoin",
                     "<em>", "</em>", "silver sparrow", "auntie ai"]
    cleaned_lower = cleaned.lower()
    if any(re.search(rf"\b{re.escape(word)}\b", cleaned_lower) for word in BLOCKED_WORDS):
        logger.info(f"TG PRE-BLOCKED → {cleaned[:120]}...")
        seen_ids.add(unique_id)
        return

    tagged = apply_smart_tagging(cleaned)

    # === IMPROVED LOW QUALITY FILTER ===
    has_tag = re.search(r"\b(JUST IN|BREAKING|ALERT|NEW):", tagged, re.IGNORECASE)
    if not has_tag:
        if is_important_crypto_or_news(tagged):
            logger.info(f"IMPORTANT CRYPTO/NEWS ALLOWED: {cleaned[:100]}...")
            if not tagged.upper().startswith(("JUST IN:", "BREAKING:", "ALERT:", "NEW:")):
                tagged = "NEW: " + tagged
        else:
            logger.info(f"TG LOW QUALITY DROPPED → {cleaned[:100]}...")
            seen_ids.add(unique_id)
            return

    # === AI Rewrite (Only short messages reach here) ===
    final_caption = await ai_rewrite(tagged)

    # === POST-AI CHECKS ===
    if final_caption is None or len(final_caption) < 20:
        logger.info(f"⛔ AI failed or empty → skipped")
        seen_ids.add(unique_id)
        return

    final_lower = final_caption.lower()
    if any(re.search(rf"\b{re.escape(word)}\b", final_lower) for word in BLOCKED_WORDS):
        logger.info(f"POST-AI BLOCKED → {final_caption[:150]}...")
        seen_ids.add(unique_id)
        return

    final_caption = add_country_flags(final_caption)

    # === SEND ===
    if original_message and original_message.video:
        await client.send_file(
            DESTINATION_CHANNEL, original_message.video,
            caption=final_caption, parse_mode='md', supports_streaming=True
        )
    else:
        figures = get_matching_images(final_caption)
        if figures:
            await client.send_file(
                DESTINATION_CHANNEL, [p.as_posix() for p in figures],
                caption=final_caption, parse_mode='md'
            )
        else:
            await client.send_message(DESTINATION_CHANNEL, final_caption, parse_mode='md')

    store_message_in_file(final_caption)
    seen_ids.add(unique_id)
    with open(SEEN_CACHE_FILE, 'a', encoding='utf-8') as f:
        f.write(unique_id + '\n')

    logger.info(f"✅ FORWARDED: {final_caption[:130]}...")

            async for msg in client.iter_messages(channel, limit=TG_OLD_LIMIT):
                if msg.text:
                    unique_id = f"tg_{channel}_{msg.id}"
                    await process_and_forward_text(msg.text, unique_id)
                    await asyncio.sleep(0.8)

        except Exception as e:
            logger.warning(f"TG startup fetch failed for {channel}: {e}")


def register_tg_listeners():
    """
    Live Telegram listener using events.
    """
    if not ENABLE_TG_SCRAPING:
        return
    from telethon import events
    from telethon.tl.types import MessageMediaDocument, DocumentAttributeVideo

    @client.on(events.NewMessage(chats=TG_SOURCE_CHANNELS))
    async def handler(event):
        try:
            if not event.message:
                return

            # ==================== ANTI-FLOOD PROTECTION ====================
            if event.message.date:
                message_age = (datetime.now(timezone.utc) - event.message.date).total_seconds()
               
                if message_age > 600:  # Ignore messages older than 10 minutes
                    if not hasattr(handler, 'old_count'):
                        handler.old_count = 0
                    handler.old_count += 1
                   
                    if handler.old_count % 25 == 0:
                        logger.warning(f"Server sent very old message with ID {event.message.id}, ignoring (age: {message_age:.0f}s)")
                    return

            # Reset counter when fresh messages arrive
            if hasattr(handler, 'old_count'):
                handler.old_count = 0
            # =====================================================================

            if not event.message:
                return

            # === SPECIAL HANDLING FOR SPECIFIC CHANNELS ===
            if event.chat and event.chat.username:
                username_lower = event.chat.username.lower().strip("@")
                
                # Video-only channels (only forward videos)
                video_only_channels = {"watcherguru"}
                if username_lower in video_only_channels:
                    has_video = bool(event.message.video) or (
                        event.message.media and isinstance(event.message.media, MessageMediaDocument) and
                        any(isinstance(attr, DocumentAttributeVideo) or
                            (getattr(event.message.media.document, 'mime_type', '') or '').startswith("video/")
                            for attr in getattr(event.message.media.document, 'attributes', []))
                    )
                    if not has_video:
                        logger.info(f"Blocked non-video message from @{event.chat.username}")
                        return
                
                # Channels where we forward ALL content
                elif username_lower in {"theblockbeats", "newslivestream"}:
                    logger.info(f"Exception applied: forwarding all content from @{event.chat.username} (ID: {event.message.id})")

            # === Original text extraction ===
            raw_text = (
                event.message.raw_text
                or event.message.text
                or event.message.message
                or ""
            )
            if event.message.media and event.message.message:
                raw_text = event.message.message

            unique_id = f"tg_{event.chat_id}_{event.message.id}"
            await process_and_forward_text(raw_text, unique_id, original_message=event.message)

        except Exception as e:
            logger.error(f"TG live handler error: {e}", exc_info=True)
