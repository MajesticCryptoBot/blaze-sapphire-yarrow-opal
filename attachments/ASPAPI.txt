def clean_message(text: str) -> str:
    if not text:
        return ""
    text = re.sub(r"https?://\S+|www\.\S+|@\w+|#\w+", "", text)
    kill_triggers = [
        r"BlockBeats", r"According\s+to\s+BlockBeats", r"According\s+to",
        r"Source\s*:\s*BlockBeats", r"via\s+BlockBeats",
    ]
    for trigger in kill_triggers:
        match = re.search(trigger, text, flags=re.IGNORECASE)
        if match:
            text = text[:match.start()].strip()
            break
    lines = []
    for line in text.split('\n'):
        line = line.strip()
        lower = line.lower()
        if any(bad in lower for bad in [
            "blockbeats", "bloomberg", "according to", "source:", "via ",
            "thea monitoring", "auntie ai", "on november", "on december"
        ]):
            continue
        if line:
            lines.append(line)
    text = "\n".join(lines)
    for w in ["Subscribe", "Follow"]:
        text = re.sub(rf"\b{w}\b", "", text, flags=re.IGNORECASE)
    text = re.sub(r"\s+", " ", text)
    text = re.sub(r"\.+$", "", text)
    return text.strip()

def is_weird_message(text: str) -> bool:
    if not text or len(text.strip()) < 8:
        return True
    stripped = text.strip()
    lower = stripped.lower()
    if re.match(r'^\s*[\d$,\.\s!]+\s*$', stripped):
        return True
    if re.search(r'now\s+\$\d+', lower) and len(stripped.split()) <= 7:
        return True
    if re.search(r'bitcoin now \$\d+', lower):
        return True
    if re.search(r'^\s*\d{4,6}\s*[!?.]*\s*$', lower):
        return True
    if len(stripped) < 25 and stripped.endswith('!'):
        return True
    return False

def apply_smart_tagging(cleaned: str) -> str:
    text = cleaned.strip()
    if not text:
        return ""
    lower = text.lower()
    crypto_assets = {"btc", "bitcoin", "eth", "ethereum", "sol", "solana", "xrp"}
    god_tier = {"musk", "elon musk", "saylor", "michael saylor", "blackrock", "microstrategy", "strategy"}
    regulatory_macro = {"fed", "powell", "jerome powell", "sec", "gensler", "gary gensler", "regulation", "regulatory", "cftc", "doj"}
    important_keywords = {
        "blackrock", "strategy", "saylor", "musk", "sec", "gensler", "fed", "powell", "gold", "jpmorgan",
        "nvidia", "tesla", "apple", "amazon", "meta", "google", "microsoft", "polymarket", "liquidated", "liquidation", "mastercard"
    }
    has_price = bool(re.search(r"(?:US\$|\$)\d{1,3}(?:[.,]\d{3})*(?:\.\d+)?|\d{1,3}(?:[.,]\d{3})*(?:\.\d+)?\s*(?:USD|USDT|BTC|ETH|dollars?)", text, re.IGNORECASE))
    def has_word(text_lower, word):
        return re.search(rf"\b{re.escape(word)}\b", text_lower, re.IGNORECASE) is not None
    has_crypto = any(has_word(lower, asset) for asset in crypto_assets)
    has_god_tier = any(has_word(lower, kw) for kw in god_tier)
    has_regulatory = any(has_word(lower, kw) for kw in regulatory_macro)
    has_important = any(has_word(lower, kw) for kw in important_keywords)
    old_news_indicators = {
        "yesterday", "last week", "last month", "on november", "on october", "on december",
        "in october", "in november", "in 2024", "in 2025", "earlier this", "previously",
        "reportedly", "according to", "announced on", "filed on", "said on", "back in"
    }
    is_old_news = any(has_word(lower, indicator) for indicator in old_news_indicators)
    urgency_words = {"now", "hits", "breaks", "surges", "drops", "falls", "jumps", "spikes", "crashes", "soars", "plunges", "crosses", "above", "below", "reclaims", "exceeds"}
    has_urgency = any(has_word(lower, word) for word in urgency_words)
    tag = ""
    if has_crypto and has_price and has_urgency and not is_old_news:
        tag = "JUST IN: "
    elif has_god_tier and not is_old_news:
        tag = "BREAKING: "
    elif has_regulatory and not is_old_news:
        tag = "ALERT: "
    elif (has_crypto or has_important) and not is_old_news:
        tag = "NEW: "
    if tag and not any(text.upper().startswith(t) for t in ["JUST IN:", "BREAKING:", "ALERT:", "NEW:"]):
        text = tag + text
    text = re.sub(r"\bUs[$&]", "US$", text, flags=re.IGNORECASE)
    text = re.sub(r"\bUS&", "US$", text)
    return text

def add_country_flags(text: str) -> str:
    if not text or not text.strip():
        return text
    lower = text.lower()
    detected_flags = set()
    country_flags = {"us": "🇺🇸", "usa": "🇺🇸", "u.s.": "🇺🇸", "america": "🇺🇸", "american": "🇺🇸", "americans": "🇺🇸", "united states": "🇺🇸", "united states of america": "🇺🇸"}
    for country, flag in country_flags.items():
        if country in {"us", "u.s.", "u.s", "usa"}:
            if re.search(r"\bu\.?s\.?\b", lower) or re.search(r"\b(united states|america)\b", lower):
                detected_flags.add(flag)
                continue
        if re.search(rf"\b{re.escape(country)}\b", lower, re.IGNORECASE):
            detected_flags.add(flag)
    if detected_flags:
        flags_str = " ".join(sorted(detected_flags)) + " "
        if any(text.upper().startswith(t) for t in ["JUST IN:", "BREAKING:", "ALERT:", "NEW:"]):
            tag_end = text.find(": ") + 2
            if tag_end > 2:
                return text[:tag_end] + flags_str + text[tag_end:]
        return flags_str + text
    return text

async def publish_to_website(message_id: int, text: str, published_at=None, photo_path=None, chat_id=None):
    """Mirror a successfully published AlphaSignalsPro post to the website.
    Website failures are isolated so Telegram publishing is never interrupted.
    """
    if not WEBSITE_INGEST_SECRET:
        logger.debug("Website ingest disabled: WEBSITE_INGEST_SECRET is not configured")
        return False
    payload = {
        "channel": WEBSITE_CHANNEL,
        "chatId": chat_id or WEBSITE_CHANNEL,
        "chatUsername": WEBSITE_CHANNEL,
        "chatTitle": WEBSITE_CHANNEL,
        "messageId": int(message_id),
        "text": text,
        "publishedAt": (published_at or datetime.now(timezone.utc)).astimezone(timezone.utc).isoformat(),
        "messageUrl": f"https://t.me/{WEBSITE_CHANNEL}/{int(message_id)}",
    }
    if photo_path:
        try:
            path = Path(photo_path)
            if path.is_file() and path.suffix.lower() in {".jpg", ".jpeg", ".png", ".webp"}:
                data = path.read_bytes()
                if len(data) <= WEBSITE_MAX_PHOTO_BYTES:
                    import base64
                    payload["photoBase64"] = base64.b64encode(data).decode("ascii")
                    payload["photoMimeType"] = {".jpg": "image/jpeg", ".jpeg": "image/jpeg", ".png": "image/png", ".webp": "image/webp"}.get(path.suffix.lower(), "image/jpeg")
                else:
                    logger.warning("Website photo skipped (>2.5 MB): %s", path.name)
        except Exception as e:
            logger.warning("Website photo preparation failed: %s", e)
    try:
        timeout = ClientTimeout(total=15, connect=5, sock_read=10)
        async with aiohttp.ClientSession(timeout=timeout) as session:
            async with session.post(
                WEBSITE_INGEST_URL,
                json=payload,
                headers={"Content-Type": "application/json", "X-Website-Ingest-Secret": WEBSITE_INGEST_SECRET},
                ssl=True,
            ) as response:
                if response.status >= 400:
                    logger.warning("Website ingest failed: HTTP %s — %s", response.status, (await response.text())[:200])
                    return False
                logger.info("Website ingest successful for Telegram message %s", message_id)
                return True
    except Exception as e:
        logger.warning("Website ingest unavailable for message %s: %s", message_id, e)
        return False

async def fetch_with_retry(session, url, params=None, max_retries=3, base_timeout=20):
    attempt = 0
    delay = 2
    while True:
        attempt += 1
        try:
            timeout = ClientTimeout(total=base_timeout + (attempt * 5), sock_connect=15, sock_read=30)
            async with session.get(url, params=params, timeout=timeout, ssl=False, headers={"User-Agent": "Mozilla/5.0 (compatible; ASPNewsBot/1.0)"}) as resp:
                if resp.status != 200:
                    raise ClientResponseError(resp.request_info, resp.history, status=resp.status)
                data = await resp.json()
                if attempt > 1:
                    logger.warning("CONNECTION RESTORED — Source: %s — Recovered after %s attempts", url, attempt)
                return data
        except (asyncio.TimeoutError, ClientConnectorError, ServerTimeoutError, OSError, ClientResponseError, ConnectionError, TimeoutError) as e:
            logger.warning("Connection/Server issue on %s (attempt %s) — %s: %s", url, attempt, type(e).__name__, str(e)[:150])
            await asyncio.sleep(delay)
            delay = min(delay * 2, 300)
        except Exception as e:
            logger.error("Unexpected error on %s (attempt %s): %s - %s", url, attempt, type(e).__name__, str(e)[:200])
            await asyncio.sleep(delay)
            delay = min(delay * 2, 300)

async def ai_rewrite(final_caption: str) -> str:
    if not final_caption.strip() or len(final_caption) < 20:
        return final_caption
    original = final_caption
    for provider in PROVIDERS:
        try:
            kwargs = {
                "model": provider["model"],
                "messages": [{"role": "system", "content": REWRITE_SYSTEM_PROMPT}, {"role": "user", "content": final_caption}],
                "temperature": 0.15, "max_tokens": 800, "top_p": 0.9,
            }
            if provider["type"] == "openai":
                kwargs["api_key"] = provider["api_key"]
                kwargs["api_base"] = provider.get("base_url")
            elif provider["type"] == "gemini":
                kwargs["api_key"] = provider["api_key"]
            response = await asyncio.to_thread(completion, **kwargs)
            rewritten = response.choices[0].message.content.strip()
            if "```" in rewritten:
                code_content = re.search(r'```(?:\w+)?\s*(.*?)\s*```', rewritten, re.DOTALL)
                if code_content:
                    rewritten = code_content.group(1).strip()
            if not rewritten or len(rewritten) < 15:
                continue
            if len(rewritten) < len(original) * 0.55:
                continue
            if re.search(r'[a-zA-Z]{3,}$', rewritten) and not rewritten.endswith(('.', '!', '?', '"', "'")):
                if len(rewritten.split()) < len(original.split()) * 0.7:
                    continue
            tag_match = re.match(r"^(JUST IN:|BREAKING:|ALERT:|NEW:)\s*", original, re.I)
            if tag_match and not rewritten.upper().startswith(tag_match.group(1).upper()):
                rewritten = tag_match.group(1) + " " + rewritten.lstrip(tag_match.group(1)).lstrip(": ").lstrip()
            logger.info(f"AI rewrite success via {provider['name']}")
            return rewritten
        except Exception as e:
            logger.warning(f"{provider['name']} failed: {str(e)[:150]}")
    logger.error("All AI providers failed → using original caption")
    return original

async def process_and_forward_text(raw_text: str, unique_id: str, original_message=None):
    global last_id
    if unique_id in seen_ids:
        return
    if not raw_text or not raw_text.strip():
        return
    cleaned = clean_message(raw_text.strip())
    if not cleaned.strip():
        return
    if is_weird_message(cleaned):
        logger.info(f"TG WEIRD BLOCKED → {cleaned[:80]}...")
        seen_ids.add(unique_id)
        return
    BLOCKED_PHRASES = [r"-\s*Link", r"-\s*链接", r"点击链接", r"详情点击", r"了解更多", r"立即查看", r"扫码关注", r"关注我们", r"广告", r"推广", r"合作咨询"]
    if any(re.search(pattern, cleaned, re.IGNORECASE) for pattern in BLOCKED_PHRASES):
        logger.info(f"TG BLOCKED PROHIBITED PHRASE → {cleaned[:120]}...")
        seen_ids.add(unique_id)
        return
    word_count = len(cleaned.split())
    text_length = len(cleaned)
    if text_length > 160 or word_count > 30:
        logger.info(f"LONG MESSAGE BLOCKED ({word_count} words, {text_length} chars) → {cleaned[:100]}...")
        seen_ids.add(unique_id)
        return
    final_caption = await ai_rewrite(cleaned)
    if final_caption is None or len(final_caption) < 20:
        seen_ids.add(unique_id)
        return
    BLOCKED_WORDS = ["buddy", "trader", "orders", "listing", "airdrop", "meme", "memecoin", "<em>", "</em>", "silver sparrow", "auntie ai"]
    final_lower = final_caption.lower()
    if any(re.search(rf"\b{re.escape(word)}\b", final_lower) for word in BLOCKED_WORDS):
        seen_ids.add(unique_id)
        return
    final_caption = apply_smart_tagging(final_caption)
    has_tag = re.search(r"\b(JUST IN|BREAKING|ALERT|NEW):", final_caption, re.IGNORECASE)
    if not has_tag:
        if is_important_crypto_or_news(final_caption):
            if not final_caption.upper().startswith(("JUST IN:", "BREAKING:", "ALERT:", "NEW:")):
                final_caption = "NEW: " + final_caption
        else:
            seen_ids.add(unique_id)
            return
    final_caption = add_country_flags(final_caption)

    # === SEND ===
    # Capture the Telegram message ID so the exact published post can be mirrored
    # to the website. Website failures are isolated and never block Telegram.
    sent_message = None
    website_photo_path = None
    if original_message and original_message.video:
        sent_message = await client.send_file(DESTINATION_CHANNEL, original_message.video, caption=final_caption, parse_mode='md', supports_streaming=True)
    else:
        figures = get_matching_images(final_caption)
        if figures:
            website_photo_path = figures[0]
            sent_message = await client.send_file(DESTINATION_CHANNEL, [p.as_posix() for p in figures], caption=final_caption, parse_mode='md')
        else:
            sent_message = await client.send_message(DESTINATION_CHANNEL, final_caption, parse_mode='md')

    try:
        website_message = sent_message[0] if isinstance(sent_message, (list, tuple)) and sent_message else sent_message
        if website_message is not None:
            await publish_to_website(
                message_id=website_message.id,
                text=final_caption,
                published_at=getattr(website_message, "date", None),
                photo_path=website_photo_path,
                chat_id=getattr(website_message, "chat_id", None),
            )
    except Exception as e:
        logger.warning("Website mirror failed but Telegram post succeeded: %s", e)

    store_message_in_file(final_caption)
    seen_ids.add(unique_id)
    with open(SEEN_CACHE_FILE, 'a', encoding='utf-8') as f:
        f.write(unique_id + '\n')
    logger.info(f"FORWARDED: {final_caption[:130]}...")

async def tg_scraper_startup_fetch():
    if not TG_FETCH_OLD_ON_START:
        return
    logger.info("TG Startup Catch-Up Enabled")
    for channel in TG_SOURCE_CHANNELS:
        try:
            async for msg in client.iter_messages(channel, limit=TG_OLD_LIMIT):
                if msg.text:
                    unique_id = f"tg_{channel}_{msg.id}"
                    await process_and_forward_text(msg.text, unique_id)
                    await asyncio.sleep(0.8)
        except Exception as e:
            logger.warning(f"TG startup fetch failed for {channel}: {e}")

def register_tg_listeners():
    if not ENABLE_TG_SCRAPING:
        return
    from telethon import events
    from telethon.tl.types import MessageMediaDocument, DocumentAttributeVideo
    @client.on(events.NewMessage(chats=TG_SOURCE_CHANNELS))
    async def handler(event):
        try:
            if not event.message:
                return
            if event.message.date:
                message_age = (datetime.now(timezone.utc) - event.message.date).total_seconds()
                if message_age > 600:
                    if not hasattr(handler, 'old_count'):
                        handler.old_count = 0
                    handler.old_count += 1
                    if handler.old_count % 25 == 0:
                        logger.warning(f"Server sent very old message with ID {event.message.id}, ignoring (age: {message_age:.0f}s)")
                    return
            if hasattr(handler, 'old_count'):
                handler.old_count = 0
            if event.chat and event.chat.username:
                username_lower = event.chat.username.lower().strip("@")
                video_only_channels = {"watcherguru"}
                if username_lower in video_only_channels:
                    has_video = bool(event.message.video) or (
                        event.message.media and isinstance(event.message.media, MessageMediaDocument) and
                        any(isinstance(attr, DocumentAttributeVideo) or (getattr(event.message.media.document, 'mime_type', '') or '').startswith("video/") for attr in getattr(event.message.media.document, 'attributes', []))
                    )
                    if not has_video:
                        logger.info(f"Blocked non-video message from @{event.chat.username}")
                        return
                elif username_lower in {"theblockbeats", "newslivestream"}:
                    logger.info(f"Exception applied: forwarding all content from @{event.chat.username} (ID: {event.message.id})")
            raw_text = event.message.raw_text or event.message.text or event.message.message or ""
            if event.message.media and event.message.message:
                raw_text = event.message.message
            unique_id = f"tg_{event.chat_id}_{event.message.id}"
            await process_and_forward_text(raw_text, unique_id, original_message=event.message)
        except Exception as e:
            logger.error(f"TG live handler error: {e}", exc_info=True)
